cmake_minimum_required(VERSION 3.24)
project(party VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PARTY_FETCH_DATACHANNEL "Fetch pinned libdatachannel and its submodules" ON)
option(PARTY_SANITIZE "Enable ASan/UBSan (Linux Clang/GCC)" OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(CURL REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
if(WIN32)
  find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
  foreach(component avcodec avformat avutil swscale swresample)
    find_library(FFMPEG_${component}_LIBRARY NAMES ${component} REQUIRED)
    list(APPEND FFMPEG_LIBRARIES ${FFMPEG_${component}_LIBRARY})
  endforeach()
  add_library(party_ffmpeg INTERFACE)
  target_include_directories(party_ffmpeg INTERFACE ${FFMPEG_INCLUDE_DIR})
  target_link_libraries(party_ffmpeg INTERFACE ${FFMPEG_LIBRARIES})
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale libswresample)
  add_library(party_ffmpeg ALIAS PkgConfig::FFMPEG)
  pkg_check_modules(CURL_PC REQUIRED IMPORTED_TARGET libcurl)
endif()
if(PARTY_FETCH_DATACHANNEL)
  include(FetchContent)
  set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
  set(NO_TESTS ON CACHE BOOL "" FORCE)
  set(USE_SYSTEM_SRTP OFF CACHE BOOL "" FORCE)
  set(LIBSRTP_TEST_APPS OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(libdatachannel GIT_REPOSITORY https://github.com/paullouisageneau/libdatachannel.git
    GIT_TAG a02b751917ac8afc8c58dc6f4461d25ff9465d48 GIT_SUBMODULES_RECURSE TRUE)
  FetchContent_MakeAvailable(libdatachannel)
else()
  find_package(LibDataChannel CONFIG REQUIRED)
endif()
add_library(party_core native/src/core.cpp native/src/delivery.cpp native/src/rtp.cpp native/src/transport.cpp native/src/signaling.cpp native/src/media.cpp native/src/player.cpp)
target_include_directories(party_core PUBLIC native/include)
target_link_libraries(party_core PUBLIC LibDataChannel::LibDataChannel nlohmann_json::nlohmann_json Threads::Threads party_ffmpeg SDL2::SDL2 CURL::libcurl)
if(NOT WIN32)
  target_link_libraries(party_core PUBLIC PkgConfig::CURL_PC)
endif()
if(MSVC)
  target_compile_options(party_core PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_compile_definitions(party_core PUBLIC NOMINMAX SDL_MAIN_HANDLED)
else()
  target_compile_options(party_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PARTY_SANITIZE AND NOT MSVC)
  target_compile_options(party_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(party_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(party native/src/main.cpp)
target_link_libraries(party PRIVATE party_core)
if(BUILD_TESTING)
  add_executable(party_unit native/tests/unit.cpp)
  target_link_libraries(party_unit PRIVATE party_core)
  add_test(NAME party_unit COMMAND party_unit)
  add_executable(party_rtp_test native/tests/rtp_test.cpp)
  target_link_libraries(party_rtp_test PRIVATE party_core)
  add_test(NAME party_rtp_test COMMAND party_rtp_test)
  add_executable(party_signaling_test native/tests/signaling_test.cpp)
  target_link_libraries(party_signaling_test PRIVATE party_core)
  add_test(NAME party_signaling_test COMMAND party_signaling_test)
  add_executable(party_loopback native/tests/loopback.cpp)
  target_link_libraries(party_loopback PRIVATE party_core)
endif()
